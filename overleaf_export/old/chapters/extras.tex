% ---------------------------------------------------------------------
%  Встроенные пометки в тексте
% ---------------------------------------------------------------------
% \term{...} — выделение нового термина (полужирный, цвет accent2)
\newcommand{\term}[1]{\textcolor{accent2}{\textbf{#1}}}
% \engterm{...} — англоязычный термин курсивом
\newcommand{\engterm}[1]{\textit{#1}}
% Короткие записи арифметики по модулю
\newcommand{\modn}[1]{\;(\mathrm{mod}\;#1)}
\newcommand{\Eul}{\varphi}
\newcommand{\Zn}[1]{\mathbb{Z}_{#1}}

% ---------------------------------------------------------------------
%  «Алгоритм» — пошаговая процедура (синяя рамка, белый фон)
% ---------------------------------------------------------------------
\newtcolorbox{algorithmbox}[1][]{enhanced, breakable,
   colback=white, colframe=accent!70!black,
   fonttitle=\bfseries\sffamily, title={Алгоритм},
   left=3mm, right=3mm, top=2mm, bottom=2mm,
   sharp corners, boxrule=0.6pt, #1}

% ---------------------------------------------------------------------
%  «Это интересно» — мягкая врезка с синим заголовком
% ---------------------------------------------------------------------
\newtcolorbox{interestbox}[1][]{enhanced, breakable,
   colback=shade, colframe=accent,
   fonttitle=\bfseries\sffamily, coltitle=white,
   left=3mm, right=3mm, top=2mm, bottom=2mm,
   sharp corners, boxrule=0.4pt,
   attach boxed title to top left={xshift=4mm, yshift=-3mm},
   boxed title style={colback=accent, sharp corners, boxrule=0pt},
   title={Это интересно}, #1}

% ---------------------------------------------------------------------
%  «Важно» — сильное предупреждение с красным заголовком
% ---------------------------------------------------------------------

   boxed title style={colback=accent2, sharp corners, boxrule=0pt},
